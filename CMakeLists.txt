cmake_minimum_required(VERSION 3.16)
project(MadDisplay LANGUAGES CXX)
add_library(MadDisplay INTERFACE)
target_include_directories(MadDisplay INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(MadDisplay INTERFACE cxx_std_14)
include(CTest)
if(BUILD_TESTING)
  foreach(name core color arduino_macro button_scale layout_model declarative showcase)
    add_executable(${name}_test tests/${name}_test.cpp)
    target_link_libraries(${name}_test PRIVATE MadDisplay)
    add_test(NAME ${name} COMMAND ${name}_test)
  endforeach()
endif()
